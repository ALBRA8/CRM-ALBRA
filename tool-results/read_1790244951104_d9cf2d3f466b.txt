    32→/* ──────────────── Demo Start Button ──────────────── */
    33→function DemoStartButton() {
    34→  const { setToken, setUser, setView } = useAppStore()
    35→  const [loading, setLoading] = useState(false)
    36→
    37→  const handleDemoStart = async () => {
    38→    setLoading(true)
    39→    try {
    40→      const data = await api.demoLogin()
    41→      const token = data.token
    42→      setToken(token)
    43→      api.setToken(token)
    44→      localStorage.setItem('crm_token', token)
    45→      setUser(data.user)
    46→      setView('dashboard')
    47→    } catch (error) {
    48→      console.error('Demo login error:', error)
    49→    } finally {
    50→      setLoading(false)
    51→    }
    52→  }
    53→
    54→  return (
    55→    <button
    56→      onClick={handleDemoStart}
    57→      disabled={loading}
    58→      className="group relative inline-flex items-center gap-3 bg-white text-emerald-700 font-bold px-10 py-5 text-lg shadow-2xl shadow-black/20 h-auto rounded-2xl transition-all duration-300 hover:scale-[1.03] hover:shadow-black/30 disabled:opacity-70 disabled:cursor-not-allowed"
    59→    >
    60→      {/* Glow effect behind */}
    61→      <div className="absolute -inset-1 bg-gradient-to-r from-emerald-400 to-teal-400 rounded-2xl opacity-30 blur-md group-hover:opacity-50 transition-opacity duration-300 -z-10" />
    62→
    63→      {loading ? (
    64→        <Loader2 className="w-6 h-6 animate-spin" />
    65→      ) : (
    66→        <Rocket className="w-6 h-6 group-hover:rotate-12 transition-transform duration-300" />
    67→      )}
    68→      <span>{loading ? 'Preparando tu demo...' : 'INICIAR'}</span>
    69→      {!loading && <ArrowRight className="w-5 h-5 group-hover:translate-x-1 transition-transform duration-300" />}
    70→    </button>
    71→  )
    72→}
