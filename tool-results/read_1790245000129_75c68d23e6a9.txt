   215→function AuditTrailSection() {
   216→  const [logs, setLogs] = useState<AuditLogEntry[]>([])
   217→  const [loading, setLoading] = useState(true)
   218→
   219→  useEffect(() => {
   220→    loadLogs()
   221→  }, [])
   222→
   223→  const loadLogs = async () => {
   224→    try {
   225→      setLoading(true)
   226→      const data = await api.getActivityLog({ limit: '10' })
   227→      setLogs((data.logs as AuditLogEntry[]) || [])
   228→    } catch {
   229→      // May fail if no auth
   230→    } finally {
   231→      setLoading(false)
   232→    }
   233→  }
   234→
   235→  if (loading) {
   236→    return (
   237→      <Card className="border-0 shadow-sm bg-white">
   238→        <CardHeader className="pb-3">
   239→          <CardTitle className="text-lg font-semibold text-slate-900">Registro de Actividad</CardTitle>
   240→        </CardHeader>
   241→        <CardContent>
   242→          <div className="space-y-3">
   243→            {[1, 2, 3].map(i => <Skeleton key={i} className="h-12 rounded-lg" />)}
   244→          </div>
   245→        </CardContent>
   246→      </Card>
   247→    )
   248→  }
   249→
   250→  return (
   251→    <motion.div
   252→      initial={{ opacity: 0, y: 12 }}
   253→      animate={{ opacity: 1, y: 0 }}
   254→      transition={{ duration: 0.3, delay: 0.48 }}
   255→    >
   256→      <Card className="border-0 shadow-sm bg-white">
   257→        <CardHeader className="pb-3">
   258→          <CardTitle className="text-lg font-semibold text-slate-900 flex items-center gap-2">
   259→            <BadgeIcon className="w-5 h-5 text-emerald-600" />
   260→            Registro de Actividad
   261→          </CardTitle>
   262→          <p className="text-xs text-slate-400">Últimas acciones en el CRM</p>
   263→        </CardHeader>
   264→        <CardContent>
   265→          {logs.length > 0 ? (
   266→            <div className="space-y-1 max-h-80 overflow-y-auto pr-1 custom-scrollbar">
   267→              {logs.map((log) => (
   268→                <div
   269→                  key={log.id}
   270→                  className="flex items-start gap-3 p-2.5 rounded-lg hover:bg-slate-50 transition-colors"
   271→                >
   272→                  <div className={`w-5 h-5 rounded-full mt-0.5 flex-shrink-0 ${auditActionColors[log.action] || 'bg-slate-400'} text-white flex items-center justify-center`}>
   273→                    {auditActionIcons[log.action] || <Activity className="w-2.5 h-2.5" />}
   274→                  </div>
   275→                  <div className="min-w-0 flex-1">
   276→                    <p className="text-sm text-slate-700 line-clamp-2">{log.description}</p>
   277→                    <div className="flex items-center gap-2 mt-0.5">
   278→                      <Badge variant="outline" className="text-[10px]">{log.action}</Badge>
   279→                      <Badge variant="outline" className="text-[10px]">{log.entity}</Badge>
   280→                      <span className="text-xs text-slate-400">
   281→                        {formatDistanceToNow(new Date(log.createdAt), { locale: es, addSuffix: true })}
   282→                      </span>
   283→                    </div>
   284→                  </div>
   285→                </div>
   286→              ))}
   287→            </div>
   288→          ) : (
   289→            <div className="py-8 text-center text-slate-400 text-sm">
   290→              <BadgeIcon className="w-10 h-10 mx-auto mb-2 text-slate-200" />
   291→              No hay registro de actividad
   292→            </div>
   293→          )}
   294→        </CardContent>
   295→      </Card>
   296→    </motion.div>
   297→  )
   298→}

   304→export function DashboardPage() {
   305→  const { setView, setSelectedClientId } = useAppStore()
   306→  const [dashboardData, setDashboardData] = useState<DashboardData | null>(null)
   307→  const [suggestions, setSuggestions] = useState<Suggestion[]>([])
   308→  const [loading, setLoading] = useState(true)
   309→  const [error, setError] = useState<string | null>(null)
   310→
   311→  useEffect(() => {
   312→    loadDashboard()
   313→  }, [])
   314→
   315→  const loadDashboard = async () => {
   316→    try {
   317→      setLoading(true)
   318→      setError(null)
   319→
   320→      const [dashboardResult, suggestionsResult] = await Promise.allSettled([
   321→        api.getDashboard(),
   322→        api.getSuggestions(),
   323→      ])
   324→
   325→      if (dashboardResult.status === 'fulfilled') {
   326→        setDashboardData(dashboardResult.value as DashboardData)
   327→      } else {
   328→        console.error('Dashboard error:', dashboardResult.reason)
   329→        setError('Error al cargar los datos del dashboard')
   330→      }
   331→
   332→      if (suggestionsResult.status === 'fulfilled') {
   333→        const sugData = suggestionsResult.value as SuggestionsData
   334→        setSuggestions(sugData.suggestions ?? [])
   335→      }
   336→      // Suggestions failure is non-critical, silently ignore
   337→    } catch {
   338→      setError('Error de conexión al cargar el dashboard')
   339→      toast.error('Error de conexión', {
   340→        description: 'No se pudieron cargar los datos del dashboard',
   341→      })
   342→    } finally {
   343→      setLoading(false)
   344→    }
   345→  }
   346→
   347→  /* ---- Loading State ---- */
   348→  if (loading) {
   349→    return (
   350→      <div className="space-y-6">
   351→        <div>
   352→          <h1 className="text-2xl font-bold text-slate-900">Dashboard</h1>
   353→          <p className="text-sm text-slate-500 mt-1">Resumen general de tu negocio</p>
   354→        </div>
   355→        <KPISkeleton />
   356→        <PipelineSkeleton />
   357→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   358→          <Card className="border-0 shadow-sm">
   359→            <CardHeader><Skeleton className="w-44 h-6 rounded" /></CardHeader>
   360→            <CardContent><Skeleton className="w-full h-64 rounded" /></CardContent>
   361→          </Card>
   362→          <Card className="border-0 shadow-sm">
   363→            <CardHeader><Skeleton className="w-36 h-6 rounded" /></CardHeader>
   364→            <CardContent><Skeleton className="w-full h-64 rounded" /></CardContent>
   365→          </Card>
   366→        </div>
   367→      </div>
   368→    )
   369→  }
   370→
   371→  /* ---- Error State ---- */
   372→  if (error || !dashboardData) {
   373→    return (
   374→      <div className="space-y-6">
   375→        <div>
   376→          <h1 className="text-2xl font-bold text-slate-900">Dashboard</h1>
   377→          <p className="text-sm text-slate-500 mt-1">Resumen general de tu negocio</p>
   378→        </div>
   379→        <Card className="border-0 shadow-sm">
   380→          <CardContent className="p-10 text-center">
   381→            <Activity className="w-12 h-12 text-slate-300 mx-auto mb-4" />
   382→            <p className="text-slate-500 font-medium">{error ?? 'No se pudieron cargar los datos'}</p>
   383→            <Button onClick={loadDashboard} variant="outline" className="mt-4">
   384→              Reintentar
   385→            </Button>
   386→          </CardContent>
   387→        </Card>
   388→      </div>
   389→    )
   390→  }
   391→
   392→  const data = dashboardData
   393→
   394→  /* ---- KPI Config ---- */
   395→  const kpis = [
   396→    {
   397→      title: 'Clientes Activos',
   398→      value: data.totalClients.toString(),
   399→      icon: Users,
   400→      trend: data.inactiveClients > 0 ? `-${data.inactiveClients} inactivos` : 'Todos activos',
   401→      trendUp: data.inactiveClients === 0,
   402→      iconBg: 'bg-emerald-50',
   403→      iconColor: 'text-emerald-600',
   404→    },
   405→    {
   406→      title: 'Valor Pipeline',
   407→      value: formatCurrency(data.pipelineValue),
   408→      icon: TrendingUp,
   409→      trend: `${data.opportunitiesByStage.reduce((a, s) => a + s.count, 0)} oportunidades`,
   410→      trendUp: true,
   411→      iconBg: 'bg-teal-50',
   412→      iconColor: 'text-teal-600',
   413→    },
   414→    {
   415→      title: 'Ingresos del Mes',
   416→      value: formatCurrency(data.monthlyRevenue),
   417→      icon: DollarSign,
   418→      trend: data.monthlyRevenue > 0 ? 'Este mes' : 'Sin ingresos',
   419→      trendUp: data.monthlyRevenue > 0,
   420→      iconBg: 'bg-green-50',
   421→      iconColor: 'text-green-600',
   422→    },
   423→    {
   424→      title: 'Tasa de Conversión',
   425→      value: `${data.conversionRate}%`,
   426→      icon: Target,
   427→      trend: data.conversionRate >= 30 ? 'Sobre promedio' : 'Mejorable',
   428→      trendUp: data.conversionRate >= 30,
   429→      iconBg: 'bg-purple-50',
   430→      iconColor: 'text-purple-600',
   431→    },
   432→  ]
   433→
   434→  /* ---- Render ---- */
   435→  return (
   436→    <div className="space-y-6">
   437→      {/* Page Title */}
   438→      <div>
   439→        <h1 className="text-2xl font-bold text-slate-900">Dashboard</h1>
   440→        <p className="text-sm text-slate-500 mt-1">Resumen general de tu negocio</p>
   441→      </div>
   442→
   443→      {/* ─── KPI Cards ─── */}
   444→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   445→        {kpis.map((kpi, index) => {
   446→          const Icon = kpi.icon
   447→          return (
   448→            <motion.div
   449→              key={kpi.title}
   450→              initial={{ opacity: 0, y: 12 }}
   451→              animate={{ opacity: 1, y: 0 }}
   452→              transition={{ duration: 0.3, delay: index * 0.06 }}
   453→            >
   454→              <Card className="border-0 shadow-sm hover:shadow-md transition-shadow bg-white">
   455→                <CardContent className="p-5">
   456→                  <div className="flex items-start justify-between">
   457→                    <div className={`w-10 h-10 rounded-lg ${kpi.iconBg} flex items-center justify-center`}>
   458→                      <Icon className={`w-5 h-5 ${kpi.iconColor}`} />
   459→                    </div>
   460→                    <div className={`flex items-center gap-1 text-xs font-medium ${kpi.trendUp ? 'text-emerald-600' : 'text-slate-400'}`}>
   461→                      {kpi.trendUp ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}

   474→      </div>
   475→
   476→      {/* ─── Pipeline Visual ─── */}
   477→      <motion.div
   478→        initial={{ opacity: 0, y: 12 }}
   479→        animate={{ opacity: 1, y: 0 }}
   480→        transition={{ duration: 0.3, delay: 0.24 }}
   481→      >
   482→        <Card className="border-0 shadow-sm bg-white">
   483→          <CardHeader className="pb-3">
   484→            <CardTitle className="text-lg font-semibold text-slate-900">Pipeline de Ventas</CardTitle>

   513→            </div>
   514→          </CardContent>

   516→      </motion.div>
   517→
   518→      {/* ─── Two Column Layout ─── */}
   519→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-6">
   520→        {/* Left Column (wider) - Revenue Chart */}
   521→        <motion.div
   522→          className="lg:col-span-3"
   523→          initial={{ opacity: 0, y: 12 }}
   524→          animate={{ opacity: 1, y: 0 }}
   525→          transition={{ duration: 0.3, delay: 0.3 }}
   526→        >
   527→          <Card className="border-0 shadow-sm bg-white h-full">
   528→            <CardHeader className="pb-3">

   541→                    </defs>
   542→                    <CartesianGrid strokeDasharray="3 3" stroke="#f1f5f9" vertical={false} />
