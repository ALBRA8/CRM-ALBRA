    33→    33→function DemoStartButton() {

    35→    35→  const [loading, setLoading] = useState(false)
    36→    36→
    37→    37→  const handleDemoStart = async () => {
    38→    38→    setLoading(true)
    39→    39→    try {
    40→    40→      const data = await api.demoLogin()
    41→    41→      const token = data.token
    42→    42→      setToken(token)
    43→    43→      api.setToken(token)
    44→    44→      localStorage.setItem('crm_token', token)
    45→    45→      setUser(data.user)
