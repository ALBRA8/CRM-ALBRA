   231→   231→// ─── Score Ring Component ────────────────────────────────────────────────────
   232→   232→
   233→   233→function ScoreRing({ score, size = 56 }: { score: number; size?: number }) {
   234→   234→  const strokeWidth = 4
   235→   235→  const radius = (size - strokeWidth) / 2
   236→   236→  const circumference = 2 * Math.PI * radius
   237→   237→  const offset = circumference - (score / 100) * circumference
   238→   238→
   239→   239→  const color = score >= 80 ? '#10b981' : score >= 60 ? '#14b8a6' : score >= 40 ? '#f59e0b' : score >= 20 ? '#f97316' : '#ef4444'
   240→   240→
   241→   241→  return (
   242→   242→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   243→   243→      <svg width={size} height={size} className="-rotate-90">
   244→   244→        <circle
   245→   245→          cx={size / 2}
   246→   246→          cy={size / 2}
   247→   247→          r={radius}
   248→   248→          fill="none"
   249→   249→          stroke="#f1f5f9"
   250→   250→          strokeWidth={strokeWidth}
   251→   251→        />
   252→   252→        <circle
   253→   253→          cx={size / 2}
   254→   254→          cy={size / 2}
   255→   255→          r={radius}
   256→   256→          fill="none"
   257→   257→          stroke={color}
   258→   258→          strokeWidth={strokeWidth}
   259→   259→          strokeDasharray={circumference}
   260→   260→          strokeDashoffset={offset}
   261→   261→          strokeLinecap="round"
   262→   262→          className="transition-all duration-700 ease-out"
   263→   263→        />
   264→   264→      </svg>
   265→   265→      <span className="absolute text-xs font-bold text-slate-700">{score}</span>
   266→   266→    </div>
   267→   267→  )
   268→   268→}
